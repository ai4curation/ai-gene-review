% Fig: three routes by which an annotation is made, and where each fails.
\begin{tikzpicture}[
  font=\sffamily\scriptsize, node distance=3mm and 4mm,
  st/.style={draw=black!40, rounded corners=2pt, fill=#1, align=center, text width=27mm, minimum height=9mm, inner sep=2pt},
  hd/.style={font=\sffamily\small\bfseries, align=center, text width=34mm, minimum height=9mm, anchor=north},
  risk/.style={align=left, text width=30mm, font=\sffamily\tiny, text=red!55!black},
  arr/.style={-{Stealth[length=1.8mm]}, draw=black!55, thick}]
% experimental
\node[hd] (h1) {Experimental\\[-1pt]{\tiny\normalfont IDA IMP IPI IGI IEP}};
\node[st=blue!6, below=of h1] (e1) {Paper reports an assay};
\node[st=blue!6, below=of e1] (e2) {Curator reads it, picks a term};
\node[st=blue!12, below=of e2] (e3) {One row: this gene, this paper};
\node[risk, below=2mm of e3] (r1) {Fails when the readout is far from the activity: expression or a distal phenotype read as participation.};
% phylogenetic
\node[hd, right=12mm of h1] (h2) {Phylogenetic\\[-1pt]{\tiny\normalfont IBD $\rightarrow$ IBA}};
\node[st=green!8, below=of h2] (p1) {Experimental rows across a PANTHER tree};
\node[st=green!8, below=of p1] (p2) {PAINT curator places function at a node};
\node[st=green!15, below=of p2] (p3) {IBA on every descendant};
\node[risk, below=2mm of p3] (r2) {Fails when the node is too deep, a subfamily diverged, or the seed was wrong.};
% electronic
\node[hd, right=12mm of h2] (h3) {Electronic\\[-1pt]{\tiny\normalfont IEA}};
\node[st=orange!10, below=of h3] (i1) {Domain, keyword, rule or ortholog};
\node[st=orange!10, below=of i1] (i2) {Mapping file links it to a term};
\node[st=orange!18, below=of i2] (i3) {IEA on every match, every release};
\node[risk, below=2mm of i3] (r3) {Fails when a term true of some members is given to all, or context is copied across species.};
\foreach \a/\b in {e1/e2,e2/e3,p1/p2,p2/p3,i1/i2,i2/i3} \draw[arr] (\a) -- (\b);
\draw[arr, dashed, draw=green!45!black] (e3.east) to[out=0, in=180] node[above, font=\sffamily\tiny, sloped]{seeds} (p1.west);
\end{tikzpicture}
